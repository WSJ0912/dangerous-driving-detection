# 智慧驭行驾驶行为检测系统 - 运行依赖
#
# 实测环境：Windows + Python 3.13.5（Anaconda），2026-10 验证可用
# 安装：  python -m pip install -r requirements.txt
#
# 版本说明：下列版本均在本机实测通过；如需放宽，可按注释中的下界使用 >=，
#           但不保证与当前代码 100% 兼容。

Flask==3.1.0              # Web 框架：路由、静态页面服务、JSON 接口
flask-cors==6.0.2         # 跨域支持（CORS），便于本机/局域网调试前端
numpy==2.1.3              # 数值计算：图像数组、坐标运算
opencv-python==4.13.0.92  # 图像解码/编码与绘制（提供 cv2 模块）
torch==2.10.0             # 深度学习运行时（CPU 版即可；有 CUDA 会自动用 GPU）
ultralytics==8.4.21       # YOLO11 模型加载与推理

# 说明：
# 1. 不要混装提供同名 cv2 模块的「无头版」OpenCV 包：两个发行包会互相覆盖，导致 cv2 不可用。
# 2. 本项目不依赖 Linux 专用的 WSGI 服务器进程管理器（代码使用 Flask 开发服务器 app.run）；
#    如需生产部署请自行引入 WSGI 服务器。
# 3. CPU-only 环境可先装精简版 PyTorch 以节省下载体积：
#    pip install torch==2.10.0 --index-url https://download.pytorch.org/whl/cpu
# 4. 测试依赖见 requirements-dev.txt。
